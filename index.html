import os import openai from moviepy.editor import * from transformers import pipeline # ========== CONFIGURAÇÕES ========== openai.api_key = "SUA_CHAVE_OPENAI_AQUI" # Substitua pela sua chave # Pasta para salvar arquivos os.makedirs("audio", exist_ok=True) os.makedirs("video", exist_ok=True) os.makedirs("assets", exist_ok=True) # ========== PASSO 1: GERAÇÃO DE ROTEIRO ========== def gerar_roteiro(prompt): print("🧠 Gerando roteiro com OpenAI...") response = openai.ChatCompletion.create( model="gpt-4", messages=[ {"role": "system", "content": "Você é um criador de roteiros de vídeos."}, {"role": "user", "content": prompt} ] ) roteiro = response.choices[0].message.content.strip() print("✅ Roteiro criado:") print(roteiro) return roteiro # ========== PASSO 2: GERAÇÃO DE VOZ ========== def gerar_audio(roteiro, nome_arquivo="audio/voz.mp3"): print("🔊 Gerando narração com Hugging Face TTS...") synthesizer = pipeline("text-to-speech", model="facebook/fastspeech2-en-ljspeech") audio = synthesizer(roteiro) with open(nome_arquivo, "wb") as f: f.write(audio["audio"]) print(f"✅ Áudio salvo em {nome_arquivo}") return nome_arquivo # ========== PASSO 3: CRIAÇÃO DO VÍDEO ========== def criar_video_com_audio(audio_path, roteiro, output="video/final.mp4"): print("🎬 Criando vídeo com MoviePy...") # Cria um fundo com texto fundo = ColorClip(size=(1280, 720), color=(0, 0, 0), duration=20) txt_clip = TextClip(roteiro[:300], fontsize=40, color='white', size=(1200, None), method="caption") txt_clip = txt_clip.set_position("center").set_duration(20) # Áudio audio = AudioFileClip(audio_path) fundo = fundo.set_audio(audio).set_duration(audio.duration) video_final = CompositeVideoClip([fundo, txt_clip]) video_final.write_videofile(output, fps=24) print(f"✅ Vídeo salvo em {output}") # ========== EXECUÇÃO ========== if __name__ == "__main__": pedido = input("Digite o pedido do vídeo: ") roteiro = gerar_roteiro(pedido) audio_path = gerar_audio(roteiro) criar_video_com_audio(audio_path, roteiro)
